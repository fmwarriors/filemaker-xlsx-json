UTL_LeerExcelJSON_sp — referencia legible de pasos; no es formato de pegado de FileMaker.

# ▼ ========================================================================== ▼
#    Propósito:      Seleccionar y leer un XLSX como JSON sin crear ni modificar registros.
#    Llamado desde:  Menú Guiones o un guion que recoge Get ( ScriptResult ). Área UTL: utilidades.
#    Entorno:        FileMaker Pro 19.5+ Mac/Windows y BaseElements; cualquier presentación.
#    Parámetros (JSON): Opcional {ruta: texto}; ruta nativa absoluta. Sin ruta abre selector.
#    Devuelve:       {errorCode,message,data:{hojas:[{nombre,columnas,filas}]}} y $$_EXCEL.RESULTADO_JSON.
# 
# 
# UTL_LeerExcelJSON_sp: sin parámetro abre selector. Parámetro JSON opcional: {ruta: ruta absoluta nativa al XLSX}.
# Resultado: {errorCode:0,message:vacio,data:{hojas:[{nombre,columnas:{A:encabezado},filas:[{numero,valores:{A:texto},tipos:{A:tipoOpenXML},estilos:{A:indice},formulas:{A:formula}}]}]}}.
# Primera fila con contenido de cada hoja = cabecera. Claves de columna A,B,...; faltantes se omiten; valores almacenados como texto; fechas numéricas sin convertir; fórmulas sin recalcular, se usa caché. Hojas de gráficos se omiten.
# Recoger Get ( ScriptResult ) inmediatamente tras Perform Script. También reemplaza $$_EXCEL.RESULTADO_JSON al inicio/final, incluso ante cancelación (errorCode=1) o error; no usar datos si errorCode<>0.
# No crea/modifica registros ni cambia presentación. Descomprime en subcarpeta temporal UUID y la elimina al finalizar.
# Las llamadas de BaseElements son Evaluate de código fijo, resuelto en el cliente. No requiere plugin en el CLI para almacenar el guion.
# 
# 
# ▲ ========================================================================== ▲
#

End If [  ]
Set Variable [ name: $_textos ; value: "" ]

Set Variable [ name: $_textos ; value: Evaluate ( "BE_FileReadText ( $_temporal & \"xl/sharedStrings.xml\" )" ) ]

Set Variable [ name: $_error ; value: Let ( errorFM = Get ( LastError ) ; Case ( $_error ; $_error ; errorFM ; errorFM ; Evaluate ( "BE_GetLastError" ) ) ) ]
If [ condition: $_error <> 0 or GetAsNumber ( $_numeroHojas ) < 1 ]
Set Variable [ name: $_error ; value: Case ( $_error ; $_error ; 10004 ) ]
Set Variable [ name: $_mensaje ; value: "El libro no contiene hojas legibles." ]
Exit Loop If [ condition: 1 ]
End If [  ]
Set Variable [ name: $_inicio ; value: Get ( CurrentTimeStamp ) ]
Set Variable [ name: $_indiceHoja ; value: 1 ]
Loop [ flush: always ]
Exit Loop If [ condition: Let ( $_error = Case ( Get ( CurrentTimeStamp ) - $_inicio > 30 ; 10011 ; $_error ) ; $_indiceHoja > $_numeroHojas or $_error <> 0 ) ]
Set Variable [ name: $_baseHoja ; value: "/*[local-name()='workbook']/*[local-name()='sheets']/*[local-name()='sheet'][" & $_indiceHoja & "]" ]
Set Variable [ name: $_xpath ; value: "string(" & $_baseHoja & "/@name)" ]
Set Variable [ name: $_nombreHoja ; value: Evaluate ( "BE_XPath ( $_libro ; $_xpath )" ) ]
Set Variable [ name: $_xpath ; value: "string(" & $_baseHoja & "/@*[local-name()='id'])" ]
Set Variable [ name: $_idHoja ; value: Evaluate ( "BE_XPath ( $_libro ; $_xpath )" ) ]
If [ condition: IsEmpty ( $_idHoja ) or Filter ( $_idHoja ; "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789_-" ) <> $_idHoja ]
Set Variable [ name: $_error ; value: 10005 ]
Set Variable [ name: $_mensaje ; value: "Identificador de hoja no compatible." ]
Exit Loop If [ condition: 1 ]
End If [  ]
Set Variable [ name: $_baseRelacion ; value: "/*[local-name()='Relationships']/*[local-name()='Relationship'][@Id='" & $_idHoja & "']" ]
Set Variable [ name: $_xpath ; value: "string(" & $_baseRelacion & "/@Type)" ]
Set Variable [ name: $_tipoHoja ; value: Evaluate ( "BE_XPath ( $_relaciones ; $_xpath )" ) ]
If [ condition: Right ( $_tipoHoja ; 10 ) = "/worksheet" ]
Set Variable [ name: $_xpath ; value: "string(" & $_baseRelacion & "/@Target)" ]
Set Variable [ name: $_destino ; value: Evaluate ( "BE_XPath ( $_relaciones ; $_xpath )" ) ]
Set Variable [ name: $_xpath ; value: "string(" & $_baseRelacion & "/@TargetMode)" ]
Set Variable [ name: $_modo ; value: Evaluate ( "BE_XPath ( $_relaciones ; $_xpath )" ) ]
If [ condition: IsEmpty ( $_destino ) or $_modo = "External" or PatternCount ( $_destino ; ".." ) or PatternCount ( $_destino ; ":" ) ]
Set Variable [ name: $_error ; value: 10006 ]
Set Variable [ name: $_mensaje ; value: "Ruta de hoja externa o no compatible." ]
Exit Loop If [ condition: 1 ]
End If [  ]
Set Variable [ name: $_archivoHoja ; value: $_temporal & Case ( Left ( $_destino ; 1 ) = "/" ; Middle ( $_destino ; 2 ; Length ( $_destino ) ) ; "xl/" & $_destino ) ]
Set Variable [ name: $_xmlHoja ; value: Evaluate ( "BE_FileReadText ( $_archivoHoja )" ) ]
Set Variable [ name: $_error ; value: Let ( errorFM = Get ( LastError ) ; Case ( errorFM ; errorFM ; Evaluate ( "BE_GetLastError" ) ) ) ]
If [ condition: $_error <> 0 or IsEmpty ( $_xmlHoja ) ]
Set Variable [ name: $_error ; value: Case ( $_error ; $_error ; 10007 ) ]
Set Variable [ name: $_mensaje ; value: "No se pudo leer la hoja: " & $_nombreHoja ]

End If [  ]
# Primera fila con contenido = encabezados. Se omiten filas sin celdas con valor/fórmula. Claves A,B,... conservan columnas vacías o duplicadas.
# Normalizar solo la copia en memoria: los atributos de formato x14ac/xr de Excel heredan namespaces que BE_XPath no incluye al extraer una fila. Se conservan contenido y atributos de datos sin prefijo; no se altera el XLSX.
Set Variable [ name: $_xsltDatos ; value: "<xsl:stylesheet version='1.0' xmlns:xsl='http://www.w3.org/1999/XSL/Transform'><xsl:output method='xml' encoding='UTF-8' omit-xml-declaration='yes'/><xsl:template match='*'><xsl:element name='{local-name()}'><xsl:apply-templates select='@*[namespace-uri() = &quot;&quot;] | node()'/></xsl:element></xsl:template><xsl:template match='@* | text()'><xsl:copy/></xsl:template><xsl:template match='comment() | processing-instruction()'/></xsl:stylesheet>" ]
Set Variable [ name: $_xmlHoja ; value: Evaluate ( "BE_XSLT_ApplyInMemory ( $_xmlHoja ; $_xsltDatos )" ) ]
Set Variable [ name: $_baseFilas ; value: "/*[local-name()='worksheet']/*[local-name()='sheetData']/*[local-name()='row'][*[local-name()='c']/*[local-name()='v' or local-name()='is' or local-name()='f']]" ]
Set Variable [ name: $_xpath ; value: "count(" & $_baseFilas & ")" ]
Set Variable [ name: $_numeroFilas ; value: Let ( [ bruto = Evaluate ( "BE_XPath ( $_xmlHoja ; $_xpath )" ) ; numero = JSONGetElement ( "[" & bruto & "]" ; 0 ) ; $_error = Case ( Left ( numero ; 1 ) = "?" or IsEmpty ( numero ) ; 10010 ; $_error ) ] ; Case ( $_error ; 0 ; numero ) ) ]
Set Variable [ name: $_error ; value: Let ( errorFM = Get ( LastError ) ; Case ( $_error ; $_error ; errorFM ; errorFM ; Evaluate ( "BE_GetLastError" ) ) ) ]

Set Variable [ name: $_filas ; value: "[]" ]

Set Variable [ name: $_xpath ; value: $_baseFilas & "[" & $_indiceFila & "]" ]
Set Variable [ name: $_xmlFila ; value: Evaluate ( "BE_XPath ( $_xmlHoja ; $_xpath ; \"\" ; 0 )" ) ]
Set Variable [ name: $_numeroFila ; value: Evaluate ( "BE_XPath ( $_xmlFila ; \"string(/*/@r)\" )" ) ]
Set Variable [ name: $_numeroCeldas ; value: Let ( [ bruto = Evaluate ( "BE_XPath ( $_xmlFila ; \"count(/*/*[local-name()='c'])\" )" ) ; numero = JSONGetElement ( "[" & bruto & "]" ; 0 ) ; $_error = Case ( Left ( numero ; 1 ) = "?" or IsEmpty ( numero ) ; 10010 ; $_error ) ] ; Case ( $_error ; 0 ; numero ) ) ]

End If [  ]

Set Variable [ name: $_valor ; value: Evaluate ( "BE_XPath ( $_xmlCelda ; \"string(/*/*[local-name()='v'])\" )" ) ]
